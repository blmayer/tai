+# tai
+
+Terminal client for the tai agent. The Neovim plugin is [tai.nvim](https://github.com/blmayer/tai).
+
+One POSIX shell script. `jq` and `curl` do the JSON and HTTP. It reads `.tai` from the project (same file as the plugin) and keeps its own session at `.tai-cli-session.json`. The `jq` filters and prompts are read from `$XDG_DATA_HOME/tai` (default `~/.local/share/tai`), then from each directory in `$XDG_DATA_DIRS`.
+
+```
+tai  llama_cpp/gemma  orchestrator  [idle]
+you: fix the build
+CODER show the error [1]
+>
+```
+
+Output is appended to the terminal. Scroll the window to read earlier lines. The input is the `>` line. The terminal title tracks the status line. The bell rings when the agent stops.
+
+| Key | Action |
+|---|---|
+| Enter | Send the line. `y` / `n` / `s` while a command is waiting |
+| Ctrl-C | Stop the agent |
+| Ctrl-L | Redraw the prompt |
+| Ctrl-U | Clear the line |
+| Ctrl-N | Clear the session |
+| Ctrl-H | Show or hide help |
+| Ctrl-Q | Quit |
+
+An empty prompt shows `ctrl-h help` in dim text. That text is not part of the line.
+
+```
+make test
+make install
+```
+
+`make install` puts the script in `~/.local/bin` and the data files in `~/.local/share/tai` (`$(PREFIX)/share/tai` when `PREFIX` is set). A `PREFIX` of `/usr/local` or `/usr` is found through the default `XDG_DATA_DIRS`.

+# File edit / read helpers. Input is the file text.
+
+def normline:
+  gsub("\\s+"; " ") | gsub("^\\s+|\\s+$"; "");
+
+def lines:
+  if . == "" then [""] else split("\n") end;
+
+def numbered($from; $to):
+  lines as $ls
+  | [range($from; $to + 1) | "\(.)" + ": " + $ls[. - 1]]
+  | join("\n");
+
+def read_range($range):
+  lines as $ls
+  | ($ls | length) as $n
+  | if $range == null or $range == "" then numbered(1; $n)
+    else
+      (if $range == "$" then {a: (-1), b: (-1)}
+       elif $range | test("^-\\d+:\\$$") then {a: ($range | sub(":\\$$"; "") | tonumber), b: (-1)}
+       elif $range | test("^\\d+:\\$$") then {a: (($range | sub(":\\$$"; "") | tonumber) - 1), b: (-1)}
+       elif $range | test("^\\d+:\\d+$") then
+         ($range | split(":")) as $p | {a: (($p[0] | tonumber) - 1), b: (($p[1] | tonumber) - 1)}
+       elif $range | test("^\\d+$") then
+         ($range | tonumber) as $k | if $k == 0 then {a: 0, b: 0} else {a: ($k - 1), b: ($k - 1)} end
+       else null end) as $r
+      | if $r == null then "Error: Invalid range \($range)"
+        else
+          (if $r.a < 0 then $n + $r.a else $r.a end) as $a0
+          | (if $r.b < 0 then $n + $r.b else $r.b end) as $b0
+          | ($a0 + 1) as $a1
+          | (if ($b0 + 1) > $n then $n else $b0 + 1 end) as $b1
+          | if $a1 < 1 or $b1 < 0 or $a1 > $n or $a1 > $b1 then "Error: Invalid range \($range)"
+            else numbered($a1; $b1) end
+        end
+    end;
+
+def find_starts($old):
+  lines as $buf
+  | ($old | lines) as $oldl
+  | ([$buf[] | normline]) as $nb
+  | ([$oldl[] | normline]) as $no
+  | if ($no | length) == 0 then []
+    else
+      [range(0; ($nb | length) - ($no | length) + 1) as $i
+        | select(all(range(0; $no | length); . as $k | $nb[$i + $k] == $no[$k]))
+        | $i]
+    end;
+
+def splice($start; $count; $newlines):
+  lines as $ls
+  | ($ls[:$start] + $newlines + $ls[$start + $count:]) | join("\n");
+
+def edit($old; $new; $multi):
+  . as $src
+  | ($new | if . == null then "" else . end | lines) as $nl
+  | if $old == null or $old == "" then
+      {ok: true, text: ($nl + ($src | lines) | join("\n"))}
+    else
+      find_starts($old) as $hits
+      | if ($hits | length) == 0 then
+          (lines | map(normline)) as $nb
+          | ($old | lines | map(normline)) as $no
+          | (reduce $nb[] as $line (0;
+              if . >= ($no | length) then .
+              elif $line == $no[.] then . + 1 else . end)) as $best
+          | {ok: false, err: "Error: old_text was not found. Call read on this file, then edit again. Best partial match: \($best)/\($no | length) lines."}
+        else
+          (if $multi then $hits | reverse else [$hits[0]] end) as $starts
+          | reduce $starts[] as $s ($src; splice($s; ($old | lines | length); $nl))
+          | {ok: true, text: .}
+        end
+    end;

+def parse_index($index):

+def node_at($path):
+  if $path == null or ($path | length) == 0 then .

+  if ($path | length) == 0 then f

+def settle_node:
+  if ((.tasks // []) | length) == 0 then .

+  elif (.tasks | any(.status == "blocked" or .status == "error")) then .status = "blocked"
+  else .status = "ok" end;

+def settle_from($path):

+def find_from($tasks; $pre; $i):

+  else

+def shell_base($cmd):
+  [$cmd | trim | splits("\\s+")] as $w

+  if ($cmd | type) != "string" or ($cmd | trim) == "" then "missing"
+  elif $cmd | test("[><]") then "redirect"

+  else "denied" end;

+        .io.op = "tool" | .io.tool = $call | .io.status = "tools" | .io.label = .stack[-1].role | .go = null

+append_raw() {
+	printf '%s%s\n' "$1" "$2" >> "$raw"
+	raw_count=$((raw_count + 1))
+}
+
+# A blank row is stored as a single kind character. Skip when the
+# transcript is empty or already ends with one.
+blank_before() {
+	[ -s "$raw" ] || return 0
+	last=$(tail -n 1 "$raw" 2>/dev/null || true)
+	[ "${#last}" -le 1 ] && return 0
+	append_raw A ""
+}
+
+# Print one stored transcript line. The kind byte selects the color.
+# The terminal wraps and keeps the line in its scrollback.
+write_raw_line() {
+	line=$1
+	if [ -z "$line" ]; then
+		printf '\n'
+		return 0
+	fi
+	kind=${line%"${line#?}"}
+	text=${line#?}
+	case $kind in
+		U) printf '\033[1m%s\033[m\n' "$text" ;;
+		T) printf '\033[2m%s\033[m\n' "$text" ;;
+		C) printf '\033[33m%s\033[m\n' "$text" ;;
+		M) printf '\033[36m%s\033[m\n' "$text" ;;
+		*) printf '%s\n' "$text" ;;
+	esac
+}
+
+# Append raw lines that are not on screen yet.
+emit_pending() {
+	[ "${raw_count:-0}" -le "${printed:-0}" ] && return 0
+	tail -n "+$((printed + 1))" "$raw" | while IFS= read -r line || [ -n "$line" ]; do
+		write_raw_line "$line"
+	done
+	printed=$raw_count
+}
+
+# The status and prompt are the only rows redrawn. The cursor sits on live_row.
+erase_live() {
+	[ "${live:-0}" -lt 1 ] && return 0
+	printf '\r'
+	if [ "${live_row:-1}" -gt 1 ]; then
+		printf '\033[%dA' "$((live_row - 1))"
+	fi
+	printf '\033[J'
+	live=0
+	live_row=1
+}

+flush_say() {
+	jq -c '.io.say[]?' "$state" > "$cache/say.jsonl"
+	while IFS= read -r row; do
+		d=$(printf '%s' "$row" | jq -r '.d // 0')
+		k=$(printf '%s' "$row" | jq -r '.k // "A"')
+		t=$(printf '%s' "$row" | jq -r '.t // empty')
+		sp=
+		i=0
+		while [ "$i" -lt "$d" ]; do sp="  $sp"; i=$((i + 1)); done
+		if [ -z "$t" ]; then
+			append_raw "$k" ""
+		else
+			blank_before
+			printf '%s' "$t" | tr -d '\r\033' > "$cache/say.txt"
+			while IFS= read -r line || [ -n "$line" ]; do
+				append_raw "$k" "$sp$line"
+			done < "$cache/say.txt"
+		fi
+	done < "$cache/say.jsonl"
+	jq '.io.say = []' "$state" > "$state.new" && mv "$state.new" "$state"
+}

+geom() {
+	# macOS ncurses tput ignores the window size when stderr is not a tty
+	# and returns the terminfo default, 24x80. Read the size from the tty.
+	if [ -t 1 ]; then
+		sz=$(stty size </dev/tty 2>/dev/null || true)
+		rows=${sz%% *}
+		cols=${sz##* }
+		if ! [ "$rows" -gt 0 ] 2>/dev/null || ! [ "$cols" -gt 0 ] 2>/dev/null; then
+			rows=24
+			cols=80
+		fi
+	fi
+	rows=${ROWS:-$rows}
+	cols=${COLS:-$cols}
+	[ "$rows" -lt 3 ] && rows=3
+	[ "$cols" -lt 20 ] && cols=20
+}

+# --- draw ----------------------------------------------------------

+set_title() {
+	tw=tai
+	if [ -n "$prov" ]; then
+		tw="$tw $prov"
+		[ -n "$model" ] && tw="$tw/$model"
+	else
+		tw="$tw no provider"
+	fi
+	tw="$tw $role"
+	if [ "$op" = "approve" ]; then
+		tw="$tw [run?]"
+	else
+		tw="$tw [$st]"
+	fi
+	[ "$ctx" != 0 ] && [ "$ctx" != null ] && tw="$tw ctx $ctx"
+	[ "$tw" = "$titlekey" ] && return 0
+	titlekey=$tw
+	tw=$(printf '%s' "$tw" | tr -d '\000-\037\177')
+	printf '\033]2;%s\007' "$tw"
+}
+
+# One prompt line per newline, including a trailing break.
+load_input() {
+	vis=$inL$inR
+	printf '%s\034' "$vis" > "$cache/input.raw"
+	awk '
+		NR > 1 { print prev }
+		{ prev = $0 }
+		END {
+			sub(/\034$/, "", prev)
+			print prev
+		}
+	' "$cache/input.raw" > "$cache/input.lines"
+	ilines=$(wc -l < "$cache/input.lines" | tr -d ' ')
+	[ "$ilines" -lt 1 ] && ilines=1
+	cur_line=$(printf '%s' "$inL" | tr -cd '\n' | wc -c | tr -d ' ')
+	rest=$inL
+	while [ "$rest" != "${rest#*
+}" ]; do
+		rest=${rest#*
+}
+	done
+	col=${#rest}
+}
+
+fit_prompt() {
+	gap=0
+	help_rows=0
+	if [ "$rows" -ge 4 ] && [ "$ilines" -le 1 ] && [ "$help" != 1 ]; then
+		gap=1
+	fi
+	if [ "$help" = 1 ]; then
+		help_all=$(wc -l < "$cache/help.txt" | tr -d ' ')
+		roomh=$((rows - 2))
+		[ "$roomh" -lt 1 ] && roomh=1
+		help_rows=$help_all
+		[ "$help_rows" -gt "$roomh" ] && help_rows=$roomh
+	fi
+	maxp=$((rows - 1 - gap - help_rows))
+	[ "$maxp" -lt 1 ] && maxp=1
+	plines=$ilines
+	[ "$plines" -gt "$maxp" ] && plines=$maxp
+	[ "$plines" -lt 1 ] && plines=1
+	if [ "$ilines" -le "$plines" ]; then
+		in_start=0
+	else
+		in_start=$((cur_line - plines + 1))
+		[ "$in_start" -lt 0 ] && in_start=0
+	fi
+	used=$((1 + gap + plines + help_rows))
+	if [ "$used" -gt "$rows" ] && [ "$help_rows" -gt 0 ]; then
+		help_rows=$((rows - 1 - gap - plines))
+		[ "$help_rows" -lt 0 ] && help_rows=0
+	fi
+}

+# Status and the prompt. Transcript lines are printed above this block.
+draw_live() {
+	geom
+	load_input
+	fit_prompt
+	st=$(jq -r '.io.status // "idle"' "$state")
+	role=$(jq -r '.io.label // "orchestrator"' "$state")
+	ctx=$(jq -r '.last_ctx // 0' "$state")
+	prov=$(jq -r '.provider // ""' "$cfgfile")
+	model=$(jq -r '.model // ""' "$cfgfile")
+	op=$(jq -r '.io.op // "idle"' "$state")
+	set_title
+	if [ "$op" = "approve" ]; then prew=5; else prew=2; fi
+	room=$((cols - prew))
+	[ "$room" -lt 1 ] && room=1
+
+	if [ -n "$prov" ]; then
+		provmodel=$prov
+		[ -n "$model" ] && provmodel=$prov/$model
+	else
+		provmodel="no provider"
+	fi
+	plain=$(printf 'tai  %s  %s  [%s]' "$provmodel" "$role" "$st")
+	extra=
+	if [ "$ctx" != 0 ] && [ "$ctx" != null ]; then
+		extra=$(printf '  ctx %s' "$ctx")
+	fi
+	if [ $((${#plain} + ${#extra})) -gt "$cols" ]; then
+		printf '%s\n' "$(printf '%s%s' "$plain" "$extra" | cut -c "1-$cols")"
+	else
+		printf '\033[1mtai\033[m  %s  \033[1m%s\033[m  ' "$provmodel" "$role"
+		case $st in
+			idle) printf '\033[2m[idle]\033[m' ;;
+			waiting) printf '\033[36m[waiting]\033[m' ;;
+			thinking) printf '\033[35m[thinking]\033[m' ;;
+			tools) printf '\033[33m[tools]\033[m' ;;
+			throttled) printf '\033[31m[throttled]\033[m' ;;
+			*) printf '[%s]' "$st" ;;
+		esac
+		printf '%s\n' "$extra"
+	fi
+
+	if [ "$help" = 1 ] && [ "$help_rows" -gt 0 ]; then
+		sed -n "1,${help_rows}p" "$cache/help.txt" | while IFS= read -r hline || [ -n "$hline" ]; do
+			printf '\033[2m%s\033[m\n' "$(printf '%s' "$hline" | cut -c "1-$cols")"
+		done
+	fi
+	if [ "$gap" = 1 ]; then
+		printf '\n'
+	fi
+
+	exec 5< "$cache/input.lines"
+	si=0
+	while [ "$si" -lt "$in_start" ]; do
+		IFS= read -r _skip <&5 || break
+		si=$((si + 1))
+	done
+	pi=0
+	poff=0
+	while [ "$pi" -lt "$plines" ]; do
+		pline_in=
+		IFS= read -r pline_in <&5 || pline_in=
+		line_no=$((in_start + pi))
+		off=0
+		if [ "$line_no" -eq "$cur_line" ]; then
+			[ "$col" -ge "$room" ] && off=$((col - room + 1))
+			poff=$off
+		fi
+		chunk=$(printf '%s' "$pline_in" | cut -c "$((off + 1))-$((off + room))")
+		if [ "$line_no" -eq 0 ]; then
+			if [ "$op" = "approve" ]; then
+				shown=$(printf '\033[33mrun?\033[m %s' "$chunk")
+			elif [ -z "$vis" ]; then
+				hint=$(printf '%s' 'ctrl-h help' | cut -c "1-$room")
+				shown=$(printf '\033[2m>\033[m \033[2m%s\033[m' "$hint")
+			else
+				shown=$(printf '\033[2m>\033[m %s' "$chunk")
+			fi
+		else
+			shown=$(printf '%*s%s' "$prew" '' "$chunk")
+		fi
+		if [ "$pi" -eq $((plines - 1)) ]; then
+			printf '%s' "$shown"
+		else
+			printf '%s\n' "$shown"
+		fi
+		pi=$((pi + 1))
+	done
+	exec 5<&-
+
+	back=$((plines - 1 - (cur_line - in_start)))
+	[ "$back" -lt 0 ] && back=0
+	[ "$back" -gt 0 ] && printf '\033[%dA' "$back"
+	printf '\r'
+	colpos=$((prew + col - poff))
+	[ "$colpos" -gt 0 ] && printf '\033[%dC' "$colpos"
+
+	live=$((1 + help_rows + gap + plines))
+	live_row=$((1 + help_rows + gap + cur_line - in_start + 1))
+	[ "$live_row" -lt 1 ] && live_row=1
+	[ "$live_row" -gt "$live" ] && live_row=$live
+	if [ "$st" = "idle" ] || [ "$op" = "approve" ]; then
+		[ "$cursor_on" = 1 ] || printf '\033[?25h'
+		cursor_on=1
+	else
+		[ "$cursor_on" = 0 ] || printf '\033[?25l'
+		cursor_on=0
+	fi
+}

+paint() {
+	erase_live
+	emit_pending
+	draw_live

+run_read() {

+	elif [ -d "$root/$file" ]; then

+	elif [ ! -f "$root/$file" ]; then
+		printf 'Error: `%s` does not exist (project %s). List with shell: ls\n' "$file" "$root" > "$resultf"
+	elif is_bin "$root/$file"; then
+		printf 'Binary file detected. Use send_image for images.\n' > "$resultf"
+	else
+		jq -nr -L "$jqdir" --rawfile src "$root/$file" --arg range "$range" \
+			'include "fs"; $src | read_range($range)' > "$resultf"
+	fi
+	show_out T

+	else

+run_image() {
+	a=$(tool_args)
+	file=$(printf '%s' "$a" | jq -r '.file // ""')
+	prompt=$(printf '%s' "$a" | jq -r '.prompt // "image"')
+	printf 'image: %s\n%s\n' "$file" "$prompt" > "$cache/call.txt"
+	show_call "$cache/call.txt"
+	ext=$(printf '%s' "$file" | sed 's/.*\.//')
+	case $ext in
+		png) mime=image/png ;;
+		jpg|jpeg) mime=image/jpeg ;;
+		gif) mime=image/gif ;;
+		webp) mime=image/webp ;;
+		bmp) mime=image/bmp ;;
+		*) mime= ;;
+	esac
+	if bad_path "$file" || [ ! -f "$root/$file" ] || [ -z "$mime" ]; then
+		printf 'Error: unsupported or missing image %s\n' "$file" > "$resultf"
+		: > "$imagef"
+	else
+		b64=$(base64 < "$root/$file" | tr -d '\n')

+mcp_rpc() {
+	name=$1 method=$2 params=$3
+	d=$cache/mcp/$name
+	id=$(( $(cat "$d/id" 2>/dev/null || echo 0) + 1 ))
+	printf '%s\n' "$id" > "$d/id"
+	jq -n --arg m "$method" --argjson id "$id" --argjson p "$params" \
+		'{jsonrpc:"2.0", id:$id, method:$m, params:$p}' > "$d/req"
+	if [ "$(cat "$d/kind" 2>/dev/null)" = http ]; then
+		key=$(jq -r --arg n "$name" '.mcps[$n].api_key // empty' "$cfgfile")
+		if [ -n "$key" ]; then
+			curl -sS -m 30 -X POST "$(cat "$d/url")" -H "Content-Type: application/json" \
+				-H "Authorization: Bearer $key" --data-binary @"$d/req" > "$d/last" || printf '{"error":"curl"}\n' > "$d/last"
+		else
+			curl -sS -m 30 -X POST "$(cat "$d/url")" -H "Content-Type: application/json" \
+				--data-binary @"$d/req" > "$d/last" || printf '{"error":"curl"}\n' > "$d/last"
+		fi
+		return 0
+	fi
+	before=$(wc -l < "$d/out" | tr -d ' ')
+	cat "$d/req" >> "$d/msg"
+	i=0
+	while [ "$i" -lt 200 ]; do
+		if awk -v b="$before" 'NR > b && /"id"/ { f = 1 } END { exit !f }' "$d/out"; then
+			awk -v b="$before" 'NR > b' "$d/out" | jq -s --argjson id "$id" 'map(select(.id == $id)) | last // {error:"no reply"}' > "$d/last"
+			return 0
+		fi
+		sleep 0.05
+		i=$((i + 1))
+	done
+	printf '{"error":"timeout"}\n' > "$d/last"
+	return 1
+}

+mcp_connect() {
+	name=$1
+	d=$cache/mcp/$name
+	mkdir -p "$d"
+	: > "$d/out"; : > "$d/msg"; printf '0\n' > "$d/id"

+	if [ -n "$url" ]; then
+		printf 'http\n' > "$d/kind"
+		printf '%s\n' "$url" > "$d/url"
+		mcp_rpc "$name" initialize '{"protocolVersion":"2024-11-05","capabilities":{},"clientInfo":{"name":"tai","version":"0.1"}}'
+		return 0
+	fi
+	cmd=$(jq -r --arg n "$name" '.mcps[$n].command // empty' "$cfgfile")
+	[ -n "$cmd" ] || { printf 'Error: mcp %s has no command or url\n' "$name"; return 1; }
+	printf 'stdio\n' > "$d/kind"
+	jq -r --arg n "$name" '.mcps[$n].args[]?' "$cfgfile" > "$d/args"
+	(
+		set --
+		while IFS= read -r arg; do set -- "$@" "$arg"; done < "$d/args"
+		tail -n +1 -f "$d/msg" | "$cmd" "$@" > "$d/out" 2>"$d/err"
+	) &
+	printf '%s\n' "$!" > "$d/pid"
+	sleep 0.1
+	mcp_rpc "$name" initialize '{"protocolVersion":"2024-11-05","capabilities":{},"clientInfo":{"name":"tai","version":"0.1"}}'
+	printf '%s\n' '{"jsonrpc":"2.0","method":"notifications/initialized"}' >> "$d/msg"
+}

+run_mcp() {
+	a=$(tool_args)
+	action=$(printf '%s' "$a" | jq -r '.action // ""')
+	name=$(printf '%s' "$a" | jq -r '.server // ""')
+	tool=$(printf '%s' "$a" | jq -r '.tool // ""')
+	printf 'mcp: %s %s %s\n' "$action" "$name" "$tool" > "$cache/call.txt"
+	show_call "$cache/call.txt"
+	case $action in
+		status) mcp_summary; cp "$cache/mcp.txt" "$resultf" ;;
+		connect)

+				printf 'connected %s\n' "$name" > "$resultf"
+			else
+				cp "$cache/mcp.err" "$resultf"
+			fi
+			mcp_summary
+			;;
+		disconnect)
+			if [ -f "$cache/mcp/$name/pid" ]; then
+				kill "$(cat "$cache/mcp/$name/pid")" 2>/dev/null || true
+				rm -f "$cache/mcp/$name/pid"
+			fi
+			printf 'disconnected %s\n' "$name" > "$resultf"
+			mcp_summary
+			;;
+		list_tools)
+			mcp_rpc "$name" "tools/list" '{}' || true
+			jq -r '.result.tools[]? | "- \(.name): \(.description // "")"' "$cache/mcp/$name/last" > "$resultf" 2>/dev/null \
+				|| cp "$cache/mcp/$name/last" "$resultf"
+			;;
+		call)
+			if jq -e --arg n "$name" --arg t "$tool" '.mcps[$n].denylist // [] | index($t) != null' "$cfgfile" >/dev/null; then
+				printf 'Error: tool %s is denied\n' "$tool" > "$resultf"

+			fi
+			;;
+		*) printf 'Error: unknown action %s\n' "$action" > "$resultf" ;;
+	esac
+	show_out T

+# Bracketed paste ends at ESC [201~. Bytes stay in the body file.
+take_paste() {
+	paste_seq=$((paste_seq + 1))
+	body=$cache/paste.$paste_seq
+	: > "$body"
+	cr=0
+	while :; do
+		rb=
+		IFS= read -r -d '' -n 1 rb </dev/tty || exit 0
+		[ -n "$rb" ] || continue
+		bb=$(printf '%d' "'$rb" 2>/dev/null) || continue
+		if [ "$bb" = 27 ]; then
+			[ "$cr" = 1 ] && printf '\n' >> "$body"
+			cr=0
+			dd bs=5 count=1 </dev/tty 2>/dev/null > "$cache/paste.end" || true
+			printf '[201~' > "$cache/paste.mark"
+			if cmp -s "$cache/paste.end" "$cache/paste.mark"; then
+				break
+			fi
+			printf '\033' >> "$body"
+			cat "$cache/paste.end" >> "$body"
+			continue
+		fi
+		if [ "$bb" = 13 ]; then
+			cr=1
+			continue
+		fi
+		if [ "$bb" = 10 ]; then
+			printf '\n' >> "$body"
+			cr=0
+			continue
+		fi
+		if [ "$cr" = 1 ]; then
+			printf '\n' >> "$body"
+			cr=0
+		fi
+		if [ "$bb" = 9 ] || [ "$bb" -ge 32 ]; then
+			printf '%s' "$rb" >> "$body"
+		fi
+	done
+	[ "$cr" = 1 ] && printf '\n' >> "$body"
+	[ -s "$body" ] || return 0
+	printf 'P %s\n' "$paste_seq" >&8 || exit 0
+}
+
+# One blocked read on the tty. An arrow is ESC plus two bytes that are
+# already buffered, so take those with dd and no timer, as ereandel does.
+# A timer waits out the whole interval on every arrow.
+key_reader() {
+	trap - EXIT INT HUP TERM
+	LC_ALL=C
+	export LC_ALL
+	exec </dev/tty
+	paste_seq=0
+	while :; do
+		c=
+		IFS= read -r -d '' -n 1 c </dev/tty || [ -n "$c" ] || exit 0
+		[ -n "$c" ] || continue
+		b=$(printf '%d' "'$c" 2>/dev/null) || continue
+		if [ "$b" = 27 ]; then
+			# The whole sequence is already buffered. Read it with no timer.
+			seq=
+			IFS= read -r -d '' -n 1 seq </dev/tty || exit 0
+			k=
+			if [ "$seq" = '[' ]; then
+				csi=
+				n=0
+				while [ "$n" -lt 16 ]; do
+					cb=
+					IFS= read -r -d '' -n 1 cb </dev/tty || exit 0
+					[ -n "$cb" ] || exit 0
+					csi=$csi$cb
+					fb=$(printf '%d' "'$cb" 2>/dev/null) || fb=0
+					[ "$fb" -ge 64 ] && break
+					n=$((n + 1))
+				done
+				case $csi in
+					A) k=UP ;;
+					B) k=DOWN ;;
+					C) k=RIGHT ;;
+					D) k=LEFT ;;
+					5~) k=PGUP ;;
+					6~) k=PGDN ;;
+					200~) k=PASTE ;;
+					201~) k= ;;
+					# Shift-Enter is its own key. A plain Enter, even as a line feed, submits.
+					13\;2u|13\;2:*u|13\;2\;*u|27\;2\;13~|27\;2\;13u|27\;2\;10~) k=NL ;;
+					13u|13\;1u|13\;1:*u|13\;1\;*u|27\;1\;13~|27\;13~) k=13 ;;
+					*) k=ESC ;;
+				esac
+			else
+				k=ESC
+			fi
+			if [ "$k" = PASTE ]; then
+				take_paste
+				continue
+			fi
+			[ -n "$k" ] || continue
+		elif [ "$b" -ge 32 ] && [ "$b" -le 126 ]; then
+			k="C $(printf "\\$(printf '%03o' "$b")")"
+		else
+			k=$b
+		fi
+		printf '%s\n' "$k" >&8 || exit 0
+	done
+}

+do_request() {
+	jq '.io.body' "$state" > "$cache/body.json"
+	limit=$(jq -r '.context_tokens // empty' "$cfgfile")
+	if [ -n "$limit" ]; then
+		chars=$(wc -c < "$cache/body.json" | tr -d ' ')
+		est=$((chars / 4))
+		if [ "$est" -gt "$limit" ]; then
+			printf '{"error":"ERROR [context]: request is %s tokens, limit is %s."}\n' "$est" "$limit" > "$resp"
+			jq_op resp
+			return
+		fi
+	fi
+	rate_wait
+	prov=$(jq -r '.provider // ""' "$cfgfile")
+	url=$(provider_url "$prov")
+	if [ -z "$url" ]; then
+		printf '%s\n' '{"error":"ERROR [provider]: set provider and model in .tai"}' > "$resp"
+		jq_op resp
+		return
+	fi
+	key=$(api_key "$prov")
+	stream=$(jq -r 'if .stream == true and .provider != "openai_responses" then "yes" else "no" end' "$cfgfile")
+	rm -f "$cache/stop"
+	if [ -n "${TAI_FAKE_RESPONSE:-}" ]; then
+		cp "$TAI_FAKE_RESPONSE" "$resp"
+		jq_op resp
+		return
+	fi
+	out=$resp
+	[ "$stream" = yes ] && out=$cache/sse
+	nflag=
+	[ "$stream" = yes ] && nflag=-N
+	# shellcheck disable=SC2086
+	(
+		trap - EXIT INT HUP TERM
+		curlpid=
+		trap '[ -n "$curlpid" ] && kill "$curlpid" 2>/dev/null || true; printf "DONE\n" >&8' EXIT
+		trap 'exit 1' TERM
+		# shellcheck disable=SC2086
+		curl -sS $nflag -D "$cache/hdr" -o "$out" -w '%{http_code}' \
+			-X POST "$url" \
+			-H "Authorization: Bearer $key" \
+			-H "Content-Type: application/json" \
+			-H "HTTP-Referer: https://terminal.pink/tai" \
+			-H "X-Title: tai" \
+			--data-binary @"$cache/body.json" > "$cache/code" &
+		curlpid=$!
+		printf '%s\n' "$curlpid" > "$cache/curl.pid"
+		wait "$curlpid"
+	) &
+	cpid=$!
+	in_req=1
+	while IFS= read -r k <&8; do
+		[ "$k" = DONE ] && break
+		case $k in
+			3) printf x > "$cache/stop"; kill_req ;;
+			8) help=$((1 - help)); paint ;;
+			12) paint ;;
+			14) clear_session ;;
+			17) quit ;;
+		esac
+	done
+	wait "$cpid" 2>/dev/null || true
+	cpid=
+	in_req=0
+	rm -f "$cache/curl.pid"
+	if [ "$drop" = 1 ]; then
+		drop=0
+		return
+	fi
+	if [ -f "$cache/stop" ]; then
+		rm -f "$cache/stop"
+		jq_op stop
+		return
+	fi
+	code=$(cat "$cache/code" 2>/dev/null || echo 000)
+	if [ "$stream" = yes ]; then
+		fold_stream
+	fi
+	case $code in
+		2*) ;;
+		000) printf '%s\n' '{"error":"ERROR [provider]: curl failed"}' > "$resp" ;;
+		*)
+			jq -n --arg c "$code" --rawfile b "$resp" \
+				'{error: ("ERROR [HTTP " + $c + "]: " + ($b[0:400]))}' > "$cache/err.json"
+			mv "$cache/err.json" "$resp"
+			;;
+	esac
+	jq_op resp
+}

+				flush_say
+				paint

+		esac

+handle() {
+	case $k in
+		LEFT) [ -n "$inL" ] && { c=${inL#"${inL%?}"}; inL=${inL%?}; inR=$c$inR; } ;;
+		RIGHT) [ -n "$inR" ] && { c=${inR%"${inR#?}"}; inR=${inR#?}; inL=$inL$c; } ;;
+		13|10) submit ;;
+		NL) inL="$inL
+" ;;
+		127) [ -n "$inL" ] && inL=${inL%?} ;;
+		8) help=$((1 - help)) ;;
+		21) inL=; inR= ;;
+		14) clear_session ;;
+		3)
+			if [ "$(jq -r '.io.op' "$state")" != idle ]; then
+				printf x > "$cache/stop"
+			else
+				inL=; inR=
+			fi
+			;;
+		4) [ -n "$inR" ] && inR=${inR#?} ;;
+		17) quit ;;
+		P\ *)
+			id=${k#P }
+			chunk=$(cat "$cache/paste.$id"; printf x)
+			chunk=${chunk%x}
+			rm -f "$cache/paste.$id"
+			inL=$inL$chunk
+			;;
+		C\ *) inL=$inL${k#C } ;;
+		*) ;;
+	esac
+}

+cleanup() {
